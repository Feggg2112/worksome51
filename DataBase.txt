CREATE TABLE `user_info` (
  `user_info` varchar(10) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '用户id',
  `nickname` varchar(20) COLLATE utf8mb4_german2_ci NOT NULL COMMENT '用户昵称',
  `email` varchar(150) COLLATE utf8mb4_german2_ci NOT NULL COMMENT '用户邮箱',
  `password` varchar(50) COLLATE utf8mb4_german2_ci NOT NULL COMMENT '密码',
  `sex` tinyint(1) DEFAULT NULL COMMENT '0女1男2未知',
  `birthday` varchar(10) COLLATE utf8mb4_german2_ci DEFAULT NULL COMMENT '出生日期',
  `school` varchar(150) COLLATE utf8mb4_german2_ci DEFAULT NULL COMMENT '学校',
  `person_introduction` varchar(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '个人简介',
  `join_time` datetime NOT NULL COMMENT '加入的时间',
  `last_login_time` datetime DEFAULT NULL COMMENT '最后登录时间',
  `last_login_ip` varchar(15) COLLATE utf8mb4_german2_ci DEFAULT NULL COMMENT '最后登录IP',
  `status` tinyint(1) NOT NULL DEFAULT '1' COMMENT '0禁用1正常',
  `notice_info` varchar(300) COLLATE utf8mb4_german2_ci DEFAULT NULL COMMENT '空间公告',
  `total_coin_count` int NOT NULL COMMENT '硬币总数量',
  `current_coin_count` int DEFAULT NULL COMMENT '现在剩下的硬币',
  `theme` tinyint(1) DEFAULT '1' COMMENT '主题',
  PRIMARY KEY (`user_info`),
  UNIQUE KEY `idx_key_email` (`email`) USING BTREE,
  UNIQUE KEY `idx_nick_name` (`nickname`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_german2_ci;
